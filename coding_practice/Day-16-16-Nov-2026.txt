# Day 16 — Ansible Roles, DynamoDB, StatefulSet, Service Check

## Ansible — Create Role Structure

# Students must create:
# roles/
#   pathnex/
#     tasks/main.yml
#     handlers/main.yml
#     templates/
#     vars/main.yml

roles/
  pathnex/
    tasks/main.yml
      ---
      - name: Install Apache
        yum:
	  name: httpd
	  state: present

      - name: Start Apache
	service: httpd
	state: started
	enabled: yes
      
      - name: Update Apache configuration
	template:
	  src: index.html.j2
	  dest: /var/www/html/index.html
	notify: Restart httpd
  
    handlers/main.yml
      ---
      - name: Restart HTTPD
	service: 
	  name: httpd
	  state: restarted
 
    vars/main.yml
      ---
      package_name: httpd
      service_name: httpd

## Shell Script - Check Service Running

#!/bin/bash

SERVICE="sshd"

if systemctl is-active --quite #SERVICE; then
  echo "$SERVICE is running"
else
  echo "$SERVICE is NOT running"
fi

# Artifacts Promotion

## Jenkins Pipeline - Promote Artifacts

pipeline {
    agent any
    environment {
        INSTITUTE_NAME = "Pathnex"
    }
    stages {
       stage('Build') {
          steps {
	     sh 'mvn clean package'
	  }
	}
	stage('Promote') {
  	   steps {
	      inpur message: "Promote artifacts to staging?"
	      sh 'echo "Promoting artifacts for $INSTITUTE_NAME"'
	   }
        }
    }
}

## Gitlab CI - Artifacts Promotion

stages:
  - build
  - promote

build:
  stage: build
  image: maven:3.8.1-jdk-17
  script:
    - mvn clean package
  artifacts:
    paths:
      - target/*.jar
    expire_in: 1 week

promote:
  stage: promote
  image: alphine:latest
  script:
    - echo "Promoting artifacts for Pathnex"
  when: manual
