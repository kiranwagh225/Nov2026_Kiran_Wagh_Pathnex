## Terraform — Create EC2 (any instance type)

- VPC
- Subnet
- Internet gateway
- Route table
- EC2 (c5.xlarge / r5.2xlarge / r6i.4xlarge / c6i.8xlarge / c6a.12xlarge)

provider "aws" {
    region = "us-east-1"
}

#Create the VPC
resource "aws_vpc" "PathnexVPC" {
    cidr_block = "10.10.0.0/16"

    tags = {
        Name = "Pathnex-VPC"
    }
}

#Create the Subnet
resource "aws_subnet" "PathnexSubnet" {
    vpc_id                  = aws_vpc.PathnexVPC.id
    cidr_block              = "10.10.1.0/24"
    map_public_ip_on_launch = true

    tags = {
        Name = "Pathnex-Subnet"
    }
}

#Create the Internet Gateway
resource "aws_internet_gateway" "PathnexIGW" {
    vpc_id = aws_vpc.PathnexVPC.id

    tags = {
        Name = "Pathnex-IGW"
    }
}

#Create Route table
resource "aws_route_table" "PathnexRT" {
    vpc_id = aws_vpc.PathnexVPC.id

    route {
        cidr_block = "0.0.0.0/0"
        gateway_id = aws_internet_gateway.PathnexIGW.id
    }

    tags = {
        Name = "Pathnex-RouteTable"
    }
}

#Create Secuirty Group
resource "aws_security_group" "PathnexSG" {
    name   = "Pathnex-SG"
    vpc_id = aws_vpc.PathnexVPC.id

    ingress {
        from_port   = 22
        to_port     = 22
        protocol    = "tcp"
        cidr_blocks = ["0.0.0.0/0"]
    }

    egress {
        from_port   = 0 
        to_port     = 0
        cidr_blocks = ["0.0.0.0/0"]
    }

    tags = {
        Name = "Pathnex-SG"
    } 
}

#Create the EC2 instance
resource "aws_instance" "PathnexEC2 {
    ami                         = "ami-12334bfbfefeufi"  
    instance_type               = "c5.xlarge"
    subnet_id                   = aws_subnet.PathnexSubnet.id
    vpc_security_group_ids      = [aws_security_group.PathnexSG.id]
    associate_public_ip_address = true

    tags = {
        Name = "Pathnex-EC2"
    }
}
## Ansible — Install Nginx & Deploy HTML File
---
- name: Install Nginx
  hosts: all
  become: yes

  tasks:
    - name: Install Nginx
      yum:
        name: nginx
        state: present
---
- name: Deploy HTML file
  hosts: all
  beccome: yes

  tasks:
    - name: Deploy HTML File
      copy:
        dest: /usr/share/nginx/html/index.html
        content: |
            Welcome to Pathnex - Nginx Deployed Using Ansible!
      


## Kubernetes — Create Deployment + Service
apiVersion: apps/v1
kind: Deployment
metadata: 
  name: Pathnex-Deployment
spec:
  replicas: 2
  selector: 
    matchLabels:
      app: nginx
    template:
      metadata:
        labels:
          app: nginx
      spec:
        containers: 
        - name: nginx
          image: nginx:latest
          ports:
          - containerPort: 80
---
apiVersion: v1
kind: Service
metadata:
  name: Pathnex-Service
spec:
  type: NodePort
  selector: 
    app: nginx
  ports:
  - port: 80
    targetPort: 80
    nodePort: 30080

# Scheduled Pipelines

## Jenkins Pipeline — Scheduled Build
You will learn how to schedule a Jenkins pipeline using cron.
pipeline {
    agent any
    environment {
        INSTITUTE_NAME = "Pathnex"
    }
    triggers {
        cron('H 2 * * *') //every day at 2 AM
    }
    stages {
        stage('checkout') {
            steps {
                git url: 'https://github.com/Pathnex/sample-java-app.git'
            }
        }
        stage('Build') {
            steps {
                sh 'mvn clean package'
            }
        }
    }
}
